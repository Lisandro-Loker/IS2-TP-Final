 29def sanitize_decimals(obj: Any) -> Any:
 30    """Convierte objetos Decimal de DynamoDB a str recursivamente."""
 31    if isinstance(obj, list):
 32        return [sanitize_decimals(item) for item in obj]
 33    if isinstance(obj, dict):
 34        return {key: sanitize_decimals(val) for key, val in obj.items()}
 35    if isinstance(obj, Decimal):
 36        return str(obj)
 37    return obj

149class ObserverManager:
150    """Implementa el patrón Observer para gestionar clientes suscriptos."""
151
152    def __init__(self) -> None:
153        """Inicializa la lista de sockets suscriptos."""
154        self._subscribers: list[socket.socket] = []
155
156    @property
157    def subscribers(self) -> list[socket.socket]:
158        """Devuelve la lista actual de sockets suscriptos."""
159        return self._subscribers
160
161    def subscribe(self, client_socket: socket.socket) -> None:
162        """Registra un nuevo socket en la lista de observadores."""
163        if client_socket not in self._subscribers:
164            self._subscribers.append(client_socket)
165            logging.debug("Nuevo cliente suscripto al Observer.")
166
167    def unsubscribe(self, client_socket: socket.socket) -> None:
168        """Elimina un socket de la lista de observadores."""
169        if client_socket in self._subscribers:
170            self._subscribers.remove(client_socket)
171            logging.debug("Cliente eliminado del Observer.")
172
173    def notify_all(self, data: dict[str, Any]) -> None:
174        """Envía una notificación JSON a todos los clientes suscriptos."""
175        payload = (json.dumps(data) + "\n").encode("utf-8")
176        disconnected: list[socket.socket] = []
177
178        for sub_socket in list(self._subscribers):
179            try:
180                sub_socket.sendall(payload)
181            except OSError as exc:
182                logging.warning("Suscriptor desconectado: %s", exc)
183                disconnected.append(sub_socket)
184
185        for dead_socket in disconnected:
186            self.unsubscribe(dead_socket)

189class CorporateDataProxy:
190    """Implementa el patrón Proxy para coordinar auditoría, datos y avisos."""
191
192    def __init__(self, observer: ObserverManager | None = None) -> None:
193        """Inicializa el Proxy con las instancias Singleton y el Observer."""
194        self.data_singleton = CorporateDataSingleton()
195        self.log_singleton = CorporateLogSingleton()
196        self.observer = observer if observer is not None else ObserverManager()
197        self.session_id = str(uuid.uuid4())
198
199    def process_request(
200        self,
201        request: dict[str, Any],
202        client_socket: socket.socket | None = None,
203    ) -> dict[str, Any] | list[dict[str, Any]]:
204        """Procesa un requerimiento JSON aplicando auditoría y reglas."""
205        client_uuid = str(request.get("UUID", "")).strip()
206        action = str(request.get("ACTION", "")).strip().lower()
207        record_id = str(request.get("ID", request.get("id", ""))).strip()
208
209        if not client_uuid or not action:
210            return {"Error": "Faltan datos mínimos necesarios (UUID, ACTION)."}
211
212        if action == "subscribe":
213            self.log_singleton.log_action(
214                client_uuid, self.session_id, "subscribe"
215            )
216            if client_socket is not None:
217                self.observer.subscribe(client_socket)
218            return {"status": "OK", "message": "Subscribed successfully"}
219
220        if action == "get":
221            if not record_id:
222                return {"Error": "Falta el campo ID para la acción get."}
223            self.log_singleton.log_action(
224                client_uuid, self.session_id, "get", record_id
225            )
226            return self.data_singleton.get_record(record_id)
227
228        if action == "list":
229            self.log_singleton.log_action(client_uuid, self.session_id, "list")
230            return self.data_singleton.list_records()
231
232        if action == "set":
233            if not record_id:
234                return {"Error": "Falta el campo ID para la acción set."}
235            self.log_singleton.log_action(
236                client_uuid, self.session_id, "set", record_id
237            )
238            result = self.data_singleton.set_record(record_id, request)
239            if "Error" not in result:
240                self.observer.notify_all(result)
241            return result
242
243        return {"Error": f"Acción '{action}' desconocida o no soportada."}
